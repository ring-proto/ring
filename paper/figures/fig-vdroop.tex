% Measured field droop along a line vs. hop distance, with the measured-below-theory gap shaded —
% the protocol ages V slightly faster than the unit-transit law because a route also dwells in the
% RIB between refreshes (extra proper time). Data: vdroop.dat (figgen). Needs \usepgfplotslibrary{fillbetween}.
\providecommand{\figpath}{figures/}
\begin{tikzpicture}
\begin{axis}[
  width=\linewidth, height=6.2cm,
  xlabel={hop distance $\delta$ from source},
  ylabel={field value toward source},
  xmin=0, xmax=8, ymin=0, ymax=1.05,
  xtick={0,1,...,8},
  legend pos=south west, legend cell align=left,
  grid=both, grid style={black!8},
  tick align=outside, tick pos=left,
]
\addplot[name path=theory, draw=black, dashed, thick, mark=none]
  table[x=delta, y=Videal] {\figpath vdroop.dat};
\addlegendentry{$1/(1+\alpha\delta)$ (theory)}
\addplot[name path=meas, draw=red!75!black, thick, mark=square*, mark options={scale=0.85}]
  table[x=delta, y=V] {\figpath vdroop.dat};
\addlegendentry{vitality $V$ (measured)}
\addplot[red!55!black, opacity=0.18] fill between[of=theory and meas];
\addlegendentry{relay dwell (extra proper time)}
\addplot[draw=blue!70!black, thick, mark=*, mark options={scale=0.85}]
  table[x=delta, y=f] {\figpath vdroop.dat};
\addlegendentry{routing potential $f$ (measured)}
\end{axis}
\end{tikzpicture}
