% Distribution of vitality across all grid nodes, split by the action threshold phi. The left tail
% past phi is the vitality horizon — nodes still field-reachable but no longer actionable. Data:
% gridfield.dat (figgen).
\providecommand{\figpath}{figures/}
\begin{tikzpicture}
\begin{axis}[
  width=\linewidth, height=5.6cm,
  xlabel={vitality $V$ (per grid node)}, ylabel={node count},
  xmin=0.4, xmax=1.02, ymin=0, ymax=22,
  ytick={0,5,10,15,20},
  tick align=outside, tick pos=left,
  axis on top,
]
\addplot[hist={bins=20, data min=0.4, data max=1.0}, fill=blue!35, draw=blue!55!black]
  table[y=V] {\figpath gridfield.dat};
\addplot[draw=red!75!black, dashed, very thick, mark=none]
  coordinates {(0.5,0) (0.5,22)};
\node[anchor=south west, font=\scriptsize, red!60!black] at (axis cs:0.505,16)
  {$\varphi$: actionable $\to$};
\node[anchor=south east, font=\scriptsize, black!60] at (axis cs:0.495,16)
  {$\gets$ horizon};
\end{axis}
\end{tikzpicture}
